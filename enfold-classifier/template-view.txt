  <p>Identifying whether an offence is Heinous or not under the definition of offences under the JJ Act, 2015 is important, particularly in the context of &mdash;</p>
  <ol style="padding-left:20px;line-height:1.6;">
    <li>Preliminary assessment under section 15, JJ Act, 2015 for children between the ages of 16&ndash;18 who have allegedly committed heinous offences.</li>
    <li>At the time of the FIR, Rule 8, JJ Model Rules, 2016, directs the police to file an FIR for Heinous offences. In other offences, a general diary complaint can be filed.</li>
    <li>The power to apprehend shall only be exercised with regard to heinous offences, unless it is in the best interest of the child.</li>
  </ol>
  <p><strong>Definition of Heinous offence.</strong><br>
  Section 2(32), JJ Act, 2015 &mdash; &ldquo;heinous offences&rdquo; includes the offences for which the minimum punishment under the Indian Penal Code or any other law for the time being in force is imprisonment for seven years or more.</p>
  <p>This tool also takes into consideration the 2021 Amendment of the JJ Act, 2015, which clarified that if an offence has no minimum punishment or a minimum punishment of less than 7 years, it will be a serious offence and will not be heinous. (Section 2(54), JJ Act, 2015)</p>

<script>
let offencesDb = [];
let currentCase = [];
let activeStatute = "";
let selectedOffence = null;
let expandedRows = new Set();

function initClassifierApp() {
  const seedEl = document.getElementById('seed-data');
  if (seedEl && seedEl.textContent.trim()) {
    try {
      offencesDb = JSON.parse(seedEl.textContent);
    } catch(e) {
      offencesDb = [];
    }
  }

  initStatuteDropdown();
  renderCaseTable();
}

function initStatuteDropdown() {
  const statuteSelect = document.getElementById('sel-statute');
  if (!statuteSelect) return;

  const statutes = [...new Set(offencesDb.map(item => item.statute))].filter(Boolean).sort();
  statuteSelect.innerHTML = '<option value="">-- Choose Statute --</option>';
  statutes.forEach(s => {
    const opt = document.createElement('option');
    opt.value = s;
    opt.textContent = s;
    statuteSelect.appendChild(opt);
  });
}

function onStatuteChange() {
  activeStatute = document.getElementById('sel-statute').value;
  document.getElementById('inp-section').value = '';
  document.getElementById('subclause-field').style.display = 'none';
  document.getElementById('btn-add-offence').disabled = true;
  selectedOffence = null;
}

function normalizeQuery(str) {
  return str.toLowerCase().replace(/^(section|sec\.?)\s*/i, '').trim();
}

function onSectionSearch() {
  const rawQuery = document.getElementById('inp-section').value;
  const query = normalizeQuery(rawQuery);
  const combo = document.getElementById('section-combo-list');
  combo.innerHTML = '';

  if (!activeStatute) {
    combo.classList.remove('open');
    return;
  }

  // Filter sections under active statute
  const matches = offencesDb.filter(o => {
    if (o.statute !== activeStatute) return false;
    if (!query) return true; // Show all when focused without query
    return o.section.toLowerCase().includes(query) || (o.label && o.label.toLowerCase().includes(query));
  });

  // Group by unique section number
  const uniqueSecMap = new Map();
  matches.forEach(m => {
    if (!uniqueSecMap.has(m.section)) {
      uniqueSecMap.set(m.section, m);
    }
  });

  if (uniqueSecMap.size === 0) {
    combo.innerHTML = '<div class="combo-empty">No matching section found</div>';
  } else {
    uniqueSecMap.forEach((firstRecord, sec) => {
      const item = document.createElement('div');
      item.className = 'combo-item';

      const snippet = firstRecord.text ? firstRecord.text.substring(0, 75).replace(/\n/g, ' ') + '…' : '';
      item.innerHTML = `<div class="lbl">Section ${sec}</div><span class="prev">${snippet}</span>`;
      item.onmousedown = (e) => { e.preventDefault(); selectSection(sec); };
      combo.appendChild(item);
    });
  }
  combo.classList.add('open');
}

function selectSection(sec) {
  document.getElementById('inp-section').value = sec;
  document.getElementById('section-combo-list').classList.remove('open');

  const subclauses = offencesDb.filter(o => o.statute === activeStatute && o.section === sec);

  if (subclauses.length === 1 && !subclauses[0].subclause) {
    document.getElementById('subclause-field').style.display = 'none';
    selectedOffence = subclauses[0];
    document.getElementById('btn-add-offence').disabled = false;
  } else {
    const subSelect = document.getElementById('sel-subclause');
    subSelect.innerHTML = '<option value="">-- Choose Sub-clause --</option>';
    subclauses.forEach((sc) => {
      const opt = document.createElement('option');
      opt.value = sc.id;
      const statusLabel = formatStatusLabel(sc.heinousStatus);
      opt.textContent = (sc.subclause ? sc.subclause : 'Main provision') + ' — ' + statusLabel;
      subSelect.appendChild(opt);
    });
    document.getElementById('subclause-field').style.display = 'flex';
    document.getElementById('btn-add-offence').disabled = true;
  }
}

function onSubclauseChange() {
  const val = document.getElementById('sel-subclause').value;
  selectedOffence = offencesDb.find(o => o.id === val);
  document.getElementById('btn-add-offence').disabled = !selectedOffence;
}

function formatStatusLabel(status) {
  if (status === 'heinous') return 'Heinous';
  if (status === 'not_heinous') return 'Not heinous';
  if (status === 'depends') return 'Depends on other charges/facts';
  return 'Unknown / needs data';
}

function addOffenceToCase() {
  if (selectedOffence && !currentCase.some(item => item.id === selectedOffence.id)) {
    currentCase.push(selectedOffence);
    renderCaseTable();
  }
}

function removeOffence(id) {
  currentCase = currentCase.filter(item => item.id !== id);
  expandedRows.delete(id);
  renderCaseTable();
}

function clearCase() {
  currentCase = [];
  expandedRows.clear();
  renderCaseTable();
}

function toggleFullText(id) {
  if (expandedRows.has(id)) {
    expandedRows.delete(id);
  } else {
    expandedRows.add(id);
  }
  renderCaseTable();
}

function renderCaseTable() {
  const wrap = document.getElementById('case-table-wrap');
  const verdictWrap = document.getElementById('case-verdict-wrap');

  if (currentCase.length === 0) {
    wrap.innerHTML = '<div class="empty-state">No charges added yet. Select a section above to assess.</div>';
    verdictWrap.innerHTML = '';
    return;
  }

  // Count verdicts
  let heinousCount = 0;
  let notHeinousCount = 0;
  let dependsCount = 0;

  currentCase.forEach(item => {
    if (item.heinousStatus === 'heinous') heinousCount++;
    else if (item.heinousStatus === 'not_heinous') notHeinousCount++;
    else dependsCount++;
  });

  // Render Banner Verdict Stamp
  let bannerClass = 'not_heinous';
  let bannerStamp = 'NOT HEINOUS';
  let bannerHeading = 'None of the entered offences are Heinous';
  let bannerDetail = 'Preliminary assessment under Section 15 of the JJ Act is not required for these charges alone.';

  if (heinousCount > 0) {
    bannerClass = 'heinous';
    bannerStamp = 'HEINOUS';
    bannerHeading = `${heinousCount} Heinous offence(s) identified`;
    bannerDetail = 'Preliminary assessment under Section 15 of the JJ Act is applicable for children aged 16–18.';
  } else if (dependsCount > 0) {
    bannerClass = 'review';
    bannerStamp = 'REVIEW FACTS';
    bannerHeading = 'One or more charges depend on factual context or underlying offences';
    bannerDetail = 'Check the caveats, underlying charges, or previous conviction records to determine final classification.';
  }

  verdictWrap.innerHTML = `
    <div class="verdict ${bannerClass}">
      <div class="stamp">${bannerStamp}</div>
      <div class="vtext">
        <strong>${bannerHeading}</strong>
        ${bannerDetail}
      </div>
      <div class="counts">
        <div><b>${heinousCount}</b>Hein.</div>
        <div><b>${notHeinousCount}</b>Not H.</div>
        <div><b>${dependsCount}</b>Dep.</div>
      </div>
    </div>`;

  // Render Case List Table
  // Wrapped in .table-scroll so narrow screens swipe horizontally instead of
  // squashing columns or breaking into fragile stacked cards.
  let html = `<div class="scroll-hint">⇆ Swipe the table sideways to see all columns</div>`;
  html += `<div class="table-scroll"><table class="list">
    <thead>
      <tr>
        <th>Statute & Section</th>
        <th>Minimum Punishment</th>
        <th>Classification</th>
        <th></th>
      </tr>
    </thead>
    <tbody>`;

  currentCase.forEach(item => {
    const badgeClass = item.heinousStatus;
    const badgeText = formatStatusLabel(item.heinousStatus);
    const isExpanded = expandedRows.has(item.id);

    html += `<tr>
      <td class="sec-cell">
        <span class="stat">${item.label}</span>
        <span class="act">${item.statute}</span>
        ${item.text ? `<button class="expand-btn" onclick="toggleFullText('${item.id}')">${isExpanded ? 'Hide full text ▲' : 'View full text ▼'}</button>` : ''}
      </td>
      <td>${item.minPunishment ? item.minPunishment.replace(/\n/g, '<br>') : 'None'}</td>
      <td><span class="badge ${badgeClass}">${badgeText}</span></td>
      <td class="cell-remove"><button class="rowremove" title="Remove" onclick="removeOffence('${item.id}')">&times;</button></td>
    </tr>`;

    if (isExpanded && item.text) {
      html += `<tr class="fulltext-row"><td colspan="4">${item.text}</td></tr>`;
    }
  });

  html += '</tbody></table></div>';
  wrap.innerHTML = html;
}

  if (!e.target.closest('.combo')) {
    const combo = document.getElementById('section-combo-list');
    if (combo) combo.classList.remove('open');
  }
